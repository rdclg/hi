import heapq

graph = {
    "Arad": [("Sibiu", 140), ("Timisoara", 118),
             ("Zerind", 75)],
    "Sibiu": [("Arad", 140), ("Fagaras", 99),
              ("Rimnicu Vilcea", 80)],
    "Fagaras": [("Sibiu", 99), ("Bucharest", 211)],
    "Rimnicu Vilcea": [("Sibiu", 80), ("Pitesti", 97)],
    "Pitesti": [("Rimnicu Vilcea", 97),
                ("Bucharest", 101)],
    "Timisoara": [("Arad", 118)],
    "Zerind": [("Arad", 75)],
    "Bucharest": [("Fagaras", 211), ("Pitesti", 101)]
}

h = {
    "Arad": 366, "Sibiu": 253, "Timisoara": 329,
    "Zerind": 374, "Fagaras": 176,
    "Rimnicu Vilcea": 193, "Pitesti": 100,
    "Bucharest": 0
}

queue = [(366, 0, "Arad", ["Arad"])]
best = {"Arad": 0}

while queue:
    f, cost, city, path = heapq.heappop(queue)

    if city == "Bucharest":
        print("Path:", path)
        print("Total distance:", cost, "km")
        break

    if cost != best[city]:
        continue

    for next_city, distance in graph[city]:
        new_cost = cost + distance

        if new_cost < best.get(next_city, float("inf")):
            best[next_city] = new_cost
            new_f = new_cost + h[next_city]

            heapq.heappush(
                queue,
                (new_f, new_cost, next_city,
                 path + [next_city])
            )
