import heapq

start = (1, 2, 3, 4, 0, 6, 7, 5, 8)
goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

def heuristic(state):
    distance = 0
    for i, tile in enumerate(state):
        if tile != 0:
            j = goal.index(tile)
            distance += abs(i // 3 - j // 3)
            distance += abs(i % 3 - j % 3)
    return distance

queue = [(heuristic(start), 0, start, [])]
best = {start: 0}

while queue:
    f, cost, state, path = heapq.heappop(queue)

    if state == goal:
        for s in path + [state]:
            print(s[:3])
            print(s[3:6])
            print(s[6:])
            print()
        print("Total moves:", cost)
        break

    if cost != best[state]:
        continue

    zero = state.index(0)

    for move in [zero-3, zero+3, zero-1, zero+1]:
        if not (0 <= move < 9):
            continue
        if zero % 3 == 0 and move == zero-1:
            continue
        if zero % 3 == 2 and move == zero+1:
            continue

        new = list(state)
        new[zero], new[move] = new[move], new[zero]
        new = tuple(new)
        new_cost = cost + 1

        if new_cost < best.get(new, float("inf")):
            best[new] = new_cost
            heapq.heappush(
                queue,
                (new_cost + heuristic(new), new_cost,
                 new, path + [state])
            )
