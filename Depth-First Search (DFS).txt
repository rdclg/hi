def dfs(a, b, visited):
    if (a, b) in visited:
        return False

    visited.add((a, b))
    print(a, b)

    if a == 2 or b == 2:
        return True

    states = [
        (4, b), (a, 3), (0, b), (a, 0),
        (max(0, a-(3-b)), min(3, a+b)),
        (min(4, a+b), max(0, b-(4-a)))
    ]

    for x, y in states:
        if dfs(x, y, visited):
            return True

    return False
